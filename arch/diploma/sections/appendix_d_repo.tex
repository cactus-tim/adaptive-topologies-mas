\section{Состав программного репозитория}
\label{app:repo}

Состав репозитория приведен по~ссылке из~приложения~\ref{app:github}.

\begingroup
\small

\subsection*{Корневая структура}

\noindent\textbf{\code{src/atm/}} --- основной пакет (13 слоев).\quad
\textbf{\code{conf/}} --- YAML-конфигурации экспериментов и
компонентов.\quad
\textbf{\code{scripts/}} --- скрипты обслуживания.\quad
\textbf{\code{notebooks/}} --- шаблон Jupyter-ноутбука для
постобработки.\quad
\textbf{\code{pyproject.toml}} --- декларация пакета и точки
входа \code{atm}.\quad
\textbf{\code{README.md}} --- инструкция по сборке.

\subsection*{Слои пакета \code{src/atm/}}

\noindent\textbf{\code{atm.core}} --- состояние и базовые типы.\quad
\textbf{\code{atm.llm}} --- оболочка провайдеров, ценообразование,
бюджеты, \code{FakeLLM}.\quad
\textbf{\code{atm.tools}} --- реестр инструментов и Docker-песочница.\quad
\textbf{\code{atm.agents}} --- класс \code{Agent}, пять ролей и
координатор.\quad
\textbf{\code{atm.topology}} --- шесть топологий поверх протокола
\code{Topology}.\quad
\textbf{\code{atm.phases}} --- FSM фаз, маршрутизатор топологий,
защитные правила, ворота передачи состояния.\quad
\textbf{\code{atm.human}} --- HITL: \code{HumanGateway}, шлюзы,
маршрутизатор ролей, политика тайм-аутов.\quad
\textbf{\code{atm.storage}} --- ORM, асинхронные сессии PostgreSQL,
писатель Parquet, чекпойнтер, миграции \code{alembic/}.\quad
\textbf{\code{atm.observability}} --- перехватчик событий LangGraph.\quad
\textbf{\code{atm.tasks}} --- загрузчики и оценщики четырех корпусов.\quad
\textbf{\code{atm.evaluation}} --- судья качества, оракулы,
агрегатор NASA-TLX.\quad
\textbf{\code{atm.experiment}} --- схемы Pydantic, OmegaConf,
CLI на Typer.\quad
\textbf{\code{atm.analysis}} --- агрегаторы метрик, графики,
построитель оракула.

\subsection*{Конфигурации \code{conf/}}

\noindent\mbox{\textbf{\code{experiments/}}} --- конфиги E1--E4 и
подтверждающих запусков.\quad
\mbox{\textbf{\code{agents/}}}, \mbox{\textbf{\code{topology/}}},
\mbox{\textbf{\code{human/}}}, \mbox{\textbf{\code{model/}}},
\mbox{\textbf{\code{tasks/}}}, \mbox{\textbf{\code{oracle/}}},
\mbox{\textbf{\code{evaluation/}}}, \mbox{\textbf{\code{sandbox/}}}
--- параметры по соответствующим компонентам.
